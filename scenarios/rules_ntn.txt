# ====================================================================
# RFC 9675-INSPIRED DTNMA DECLARATIVE RULES: NON-TERRESTRIAL NETWORKS (NTN)
# Targeted for high-latency, asymmetric space links and orbital relays
# with explicit catch-all release when surge/hazard subsides.
# ====================================================================

# Rule 0 (Priority 100): Space Critical Congestion or Low Battery Failsafe
RULE: ARI://DTNMA/SRL/NTN_CRITICAL_FAILSAFE | PRIORITY: 100 | DESC: Low energy survival or critical telecommand saturation
IF: (EDD_ENERGY_SOC < 0.20) || (EDD_CRIT_QUEUE_RATIO >= 0.50)
THEN: CTRL_SET_SPRAY_QUOTA(1); CTRL_SET_INGRESS_MODE(2); CTRL_SET_TELEMETRY_INTERVAL(600); CTRL_SET_BACKPRESSURE(true)

# Rule 1 (Priority 80): Active Space Buffer Surge
RULE: ARI://DTNMA/SRL/NTN_SURGE_THROTTLE | PRIORITY: 80 | DESC: Rapid orbital buffer fill surge
IF: (EDD_QUEUE_RATIO >= 0.88) && (EDD_QUEUE_DRIFT >= 0.008)
THEN: CTRL_SET_SPRAY_QUOTA(2); CTRL_SET_INGRESS_MODE(1); CTRL_SET_TELEMETRY_INTERVAL(300); CTRL_SET_BACKPRESSURE(true)

# Rule 2 (Priority 50): Multi-Orbiter Dense Pass Under Buffer Pressure
RULE: ARI://DTNMA/SRL/NTN_ORBITAL_CLUSTER | PRIORITY: 50 | DESC: Congested dense orbital rendezvous cluster
IF: (EDD_QUEUE_RATIO >= 0.75) && (EDD_NODE_DENSITY >= 4.0)
THEN: CTRL_SET_SPRAY_QUOTA(4); CTRL_SET_INGRESS_MODE(0); CTRL_SET_TELEMETRY_INTERVAL(180); CTRL_SET_BACKPRESSURE(true)

# Rule 3 (Priority 10): Catch-All Deep Space Sparse / Nominal Release
RULE: ARI://DTNMA/SRL/NTN_DEEP_SPACE_SPARSE | PRIORITY: 10 | DESC: Nominal orbital and deep space exploratory regime
IF: (EDD_QUEUE_RATIO >= 0.0)
THEN: CTRL_SET_SPRAY_QUOTA(6); CTRL_SET_INGRESS_MODE(0); CTRL_SET_TELEMETRY_INTERVAL(60); CTRL_SET_BACKPRESSURE(true)
