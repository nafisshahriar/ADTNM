# ====================================================================
# RFC 9675 DTNMA DECLARATIVE RULES: VDTN (-20% THRESHOLD PERTURBATION)
# Used for RQ5 Declarative Programmability Sensitivity Analysis
# ====================================================================

RULE: ARI://DTNMA/SRL/VDTN_BUFFER_CRITICAL | PRIORITY: 100 | DESC: Critical buffer congestion or low energy (-20% thresholds)
IF: (EDD_QUEUE_RATIO >= 0.76) || (EDD_ENERGY_SOC < 0.16)
THEN: CTRL_SET_SPRAY_QUOTA(1); CTRL_SET_INGRESS_MODE(2); CTRL_SET_TELEMETRY_INTERVAL(600); CTRL_SET_BACKPRESSURE(true)

RULE: ARI://DTNMA/SRL/VDTN_TRAFFIC_JAM | PRIORITY: 60 | DESC: Congested vehicle cluster (-20% thresholds)
IF: (EDD_QUEUE_RATIO >= 0.52) && (EDD_NODE_DENSITY >= 4.0) && (EDD_NODE_SPEED < 1.6)
THEN: CTRL_SET_SPRAY_QUOTA(3); CTRL_SET_INGRESS_MODE(1); CTRL_SET_TELEMETRY_INTERVAL(300); CTRL_SET_BACKPRESSURE(true)

RULE: ARI://DTNMA/SRL/VDTN_HIGHWAY_PASS | PRIORITY: 50 | DESC: Congested high-speed highway pass (-20% thresholds)
IF: (EDD_QUEUE_RATIO >= 0.52) && (EDD_NODE_SPEED >= 8.0)
THEN: CTRL_SET_SPRAY_QUOTA(4); CTRL_SET_INGRESS_MODE(0); CTRL_SET_TELEMETRY_INTERVAL(120); CTRL_SET_BACKPRESSURE(true)

RULE: ARI://DTNMA/SRL/VDTN_URBAN_NOMINAL | PRIORITY: 10 | DESC: Uncongested nominal cruising (-20% thresholds)
IF: (EDD_QUEUE_RATIO < 0.68)
THEN: CTRL_SET_SPRAY_QUOTA(6); CTRL_SET_INGRESS_MODE(0); CTRL_SET_TELEMETRY_INTERVAL(60); CTRL_SET_BACKPRESSURE(false)
